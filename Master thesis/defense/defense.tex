\section{Methodology}
\begin{frame}{Four Predictive Formulations}
\renewcommand{\arraystretch}{1.7}
\begin{tabularx}{\textwidth}{@{}p{36mm}X p{39mm}@{}}
\toprule
\textbf{Formulation} & \textbf{Question at time $t$} & \textbf{Output}\\\midrule
\textcolor{forecast}{\textbf{Autoregressive}} & What are the next ten signal values? & Future trajectory\\
\textcolor{current}{\textbf{Current-level}} & How long until the signal falls below the threshold $S_t-0.5$ dB? & Duration estimate\\
\textcolor{longfade}{\textbf{Long-fade}} & Does the entire grouped event last at least 300 seconds? & Class probability\\
\textcolor{survival}{\textbf{Survival}} & Will the current fade remain active for more than $u$ seconds? & Survival curve $S(u\mid X_t)$\\
\bottomrule
\end{tabularx}
\takeaway{All use past signal information; their targets are not interchangeable.}
\end{frame}

\section{Models}
\begin{frame}{Autoregressive Forecasting}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\textcolor{forecast}{\textbf{Predict the next ten values}}\[
X_t=(S_{t-29},\ldots,S_t)\ \longrightarrow\ (\widehat S_{t+1},\ldots,\widehat S_{t+10})
\]
\small GRU sequence-to-vector and encoder-decoder; PatchTST; XGBoost per horizon; Chronos zero-shot.
\vspace{3mm}
\rulebox{$a_t=1\quad\text{if }\widehat S_{t+k}\geq10\ \ \forall k=1,\ldots,10$}
\vspace{3mm}
\smallnote{Raw and context-standardized trainable variants. Chronos uses raw contexts; L120 is an additional experiment.}
\column{.48\textwidth}
\textbf{External test, native L30 grid}
\vspace{3mm}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}lrr@{}}\toprule
Model & RMSE $\downarrow$ & F1 $\uparrow$\\
 & (dB) & (switch)\\\midrule
GRU S2V raw & \textbf{2.852} & 0.674\\
PatchTST raw & 3.088 & \textbf{0.808}\\\bottomrule
\end{tabular}\par
\vspace{5mm}
PatchTST provides the stronger switch decision despite a higher trajectory RMSE.
\end{columns}
\takeaway{The best forecasting score does not select the best operational policy.}
\end{frame}

% 9 -- 1:25
\begin{frame}{Current-Level Persistence}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\textcolor{current}{\textbf{Predict time below the current level}}\[
R_t^{\mathrm{level}}=\inf\{u>0:S_{t+u}<S_t-0.5\}
\]
\small Learnable shapelets with MLP, convolutional or Transformer heads; XGBoost baseline. Relative histories plus scalar signal context.
\rulebox{$a_t=1\quad\text{if }S_t\geq10\ \text{and }\widehat R_t^{\mathrm{level}}\geq300\,\mathrm{s}$}
\column{.48\textwidth}
\textbf{XGBoost: external test, native grid}
\vspace{3mm}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}lr@{}}\toprule
Duration MAE $\downarrow$ & 586.9 s\\
Duration median AE $\downarrow$ & 27.2 s\\\midrule
Switch precision $\uparrow$ & 1.000\\
Switch recall $\uparrow$ & 0.822\\
Switch F1 $\uparrow$ & 0.902\\\bottomrule
\end{tabular}
\smallnote{Shapelet variants never activate the final switch: their duration estimates do not satisfy the operational gate.}
\end{columns}
\takeaway{A conservative XGBoost policy succeeds where the shapelet duration models fail.}
\end{frame}

% 10 -- 1:20
\begin{frame}{Long-Fade Detection}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\textcolor{longfade}{\textbf{Classify the entire grouped event}}\[
p_t=P(D_{\mathrm{event}}\geq300\,\mathrm{s}\mid X_t)
\]
\small XGBoost, causal TCN and shapelet convolution classifier.
\vspace{3mm}
\rulebox{$a_t=1\quad\text{if }p_t\geq0.5$}
\smallnote{The label remains positive throughout a long grouped event, including near its end. It does not describe the remaining duration.}
\column{.48\textwidth}
\textbf{TCN: external test, native grid}
\vspace{3mm}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}lr@{}}\toprule
Classification AUPRC $\uparrow$ & $\approx1.000$\\\midrule
Switch precision $\uparrow$ & 0.540\\
Switch recall $\uparrow$ & 1.000\\
Switch F1 $\uparrow$ & 0.702\\
Active duration & 445 min\\\bottomrule
\end{tabular}
\smallnote{Classification is evaluated on an event-conditioned, strongly imbalanced domain.}
\end{columns}
\takeaway{Near-perfect event classification does not imply accurate switch timing.}
\end{frame}

% 11 -- 1:35
\begin{frame}{Survival Persistence}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\textcolor{survival}{\textbf{Predict residual-event uncertainty}}\[
S(u\mid X_t)=P(R_t>u\mid X_t)
\]
\small XGBoost-AFT uses survival bounds; the TCN predicts discrete hazards. Both retain censored samples.
\rulebox{$a_t=1\text{ if }S(300\,\mathrm{s}\mid X_t)\geq0.65$\\and $S_t\geq10$}
\smallnote{\textcolor{alertred}{0.65 is an exploratory test-selected threshold, not validation-calibrated.}}
\vspace{1mm}
\small Native test: TCN has lower IBS (0.103 vs 0.117); AFT higher switch F1 (0.863 vs 0.784).

\section{Conclusions}
\begin{frame}{Limitations and Next Steps}
\renewcommand{\arraystretch}{2}
\begin{tabularx}{\textwidth}{@{}p{59mm}X@{}}\toprule
\textbf{Experimental limitation} & \textbf{Next step}\\\midrule

% 16 -- 0:50
\begin{frame}{Conclusions}
\begin{enumerate}\setlength{\itemsep}{6mm}
\item \textbf{Task definition changes the switch behavior.}\\
\small A future trajectory, a level duration and an event label answer different questions.
\item \normalsize\textbf{Native predictive scores do not determine operational quality.}\\
\small Evaluate timing, false activation, missed activity and decision coverage.
\item \normalsize\textbf{Tabular baselines remain competitive.}\\
\small Current-level XGBoost and survival XGBoost-AFT expose useful operating trade-offs.
\end{enumerate}
\vfill
{\large\color{navy}\textbf{Evaluate the prediction and the decision policy together.}\par}
